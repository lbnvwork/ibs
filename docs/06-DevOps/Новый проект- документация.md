### **1.5. Настройка Docker Compose и конфигурации**

```YAML
services:
  db:
    image: postgres:15-alpine
    env_file: .env
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - app-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DB_USER} -d ${DB_NAME}"]
      interval: 5s
      timeout: 3s
      retries: 5

  php:
    build:
      context: docker/php
      dockerfile: Dockerfile
    env_file: .env
    volumes:
      - ./:/var/www/html
      - /var/www/html/vendor
      - /var/www/html/node_modules
    networks:
      - app-network
    depends_on:
      db:
        condition: service_healthy

  nginx:
    image: nginx:1.25-alpine
    ports:
      - "8080:80"
    volumes:
      - ./docker/nginx/nginx.conf:/etc/nginx/conf.d/default.conf
      - ./public:/var/www/html/public:ro
    networks:
      - app-network
    depends_on:
      - php

  frontend:
    image: node:18-alpine
    working_dir: /app
    volumes:
      - ./vue:/app
    environment:
      VITE_API_URL: http://nginx/api
    networks:
      - app-network

volumes:
  postgres_data:

networks:
  app-network:
    driver: bridge
```

### **Конфигурация Nginx (nginx/default.conf)**

```JSON
server {
    listen 80;

    server_name localhost;

    root /var/www/html/public;  # Путь внутри контейнера Nginx

    index index.php index.html index.htm;

    location / {
        try_files $uri $uri/ /index.php$is_args$args;
    }

    location ~ \.php$ {
        fastcgi_pass ibs-php:9000;  # Используем имя сервиса PHP
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;

        # Дополнительные параметры для Symfony
        fastcgi_param HTTPS off;
        fastcgi_param HTTP_PROXY "";
        fastcgi_buffers 16 16k;
        fastcgi_buffer_size 32k;
    }

    # Запрещаем доступ к скрытым файлам
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }
}
```
